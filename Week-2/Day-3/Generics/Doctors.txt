class Doctor
    attributes: name, specialization, experienceYears
    method toString()
        return name + " - " + specialization + " (" + experienceYears + " years)"

start
    hospitalDirectory = new Map<String, List<Doctor>>()

    hospitalDirectory["Cardiology"] = [ 
        Doctor("Dr. Mehta","Cardiologist",15), 
        Doctor("Dr. Kapoor","Cardiologist",10) 
    ]
    hospitalDirectory["Neurology"] = [ 
        Doctor("Dr. Sharma","Neurologist",10), 
        Doctor("Dr. Das","Neurologist",8) 
    ]
    hospitalDirectory["Pediatrics"] = [ 
        Doctor("Dr. Ritu","Pediatrician",12), 
        Doctor("Dr. Anjali","Pediatrician",7) 
    ]

    input = "Neurology"
    if input in hospitalDirectory
        for doctor in hospitalDirectory[input]
            print doctor.toString()
end
